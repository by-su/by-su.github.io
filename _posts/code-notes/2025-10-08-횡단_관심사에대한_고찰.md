## 도입

다음 코드에서 slug가 "slug"로 하드코딩 되어 있는데요, 이 코드를 다음 스펙에 맞게 수정해야합니다.
- 4자 slug생성 (충돌 시 5~8자 생성)
- 허용 문자만 사용: `^[a-zA-Z0-9_-]+$` 일치
- 혼동 문자 제외: `0`, `O`, `l`, `I` 미포함

```kotlin
fun createShortUrl(req: ShortUrlRequest): ShortUrlResponse? {  
    // URL 검증 로직  
    val isValid = !req.url.contains("localhost") &&  
            (req.url.length <= 2048) &&  
            (req.url.startsWith("http://") || req.url.startsWith("https://"))  
  
    if (!isValid) return null  
  
    // 비즈니스 로직 (예: DB 저장 or slug 생성)  
    return ShortUrlResponse(  
        id = 1L,  
        slug = "slug",  
        shortUrl = "testurl",  
        originalUrl = req.url,  
        createdAt = LocalDateTime.now()  
    )  
  
}
```

생각한 대안은 총 세 가지 입니다. `private function`, `유틸성 클래스`, `도메인 서비스`
- private 메서드: 재사용성/교체성/테스트 대역 주입이 어려움.
- 범용 유틸: 알파벳/길이/금칙어 등 도메인 맥락을 잃어 설계가 퍼짐.
- 도메인 서비스: URL 단축이라는 맥락(허용 문자, 길이 정책, 금칙어, 통계)을 내포해 응집도↑.

## Service -> Service로의 횡단 관심사
저는 Service에서 또 다른 Service를 호출하면 순환 참조, 비즈니스 로직 파악, 트랜잭션 관리 등의 어려움이 있어 Service -> Service로의 호출을 꺼리는 편인데요, 어떻게 보면 네이밍이 다를 뿐이지 RandomShortCodeGenerator도 서비스가 아닌가에 대한 물음이 들어 이거에 대한 답은 찾아보고자 합니다.

```kotlin
@Service
class ShortenUrlService(
    private val randomShortCodeGenerator: RandomShortCodeGenerator
) {
    fun createShortUrl(req: ShortUrlRequest): ShortUrlResponse? {
        // 비즈니스 로직 조율
        val shortCode = randomShortCodeGenerator.generate()
        return ShortUrlResponse(...)
    }
}
```

짧은 URL을 생성하는 것은 서비스의 주요 목적이므로, `ShortenUrlService`의 핵심 비즈니스 로직. 즉 도메인 로직에 속합니다.
- `RandomShortCodeGenerator`는 `@Component` (유틸리티 성격)
- 순수 계산만 수행 (외부 의존성 없음)
- 단일 책임만 가짐


실제로 문제가 되는 경우 다음과 같습니다.

### 문제 1: 트랜잭션 경계 혼란

```
@Service
class ShortenUrlService(
    private val urlRepository: UrlRepository,
    private val analyticsService: AnalyticsService
) {
    @Transactional
    fun createShortUrl(req: ShortUrlRequest): ShortUrlResponse {
        val url = urlRepository.save(UrlEntity(...))
        
        // 문제: analyticsService도 @Transactional이면?
        // - 트랜잭션 전파 방식에 따라 예상치 못한 동작
        // - analyticsService 실패 시 url 저장도 롤백될 수 있음
        analyticsService.recordCreation(url.id)  // ← 트랜잭션 경계 혼란
        
        return ShortUrlResponse(...)
    }
}

@Service
class AnalyticsService(
    private val analyticsRepository: AnalyticsRepository
) {
    @Transactional  // ← 여기도 트랜잭션!
    fun recordCreation(urlId: Long) {
        analyticsRepository.save(AnalyticsEvent(...))
    }
}
```

- 트랜잭션 경계가 불명확
- Analytics 실패가 URL 생성을 롤백시킬 수 있음
- 테스트가 복잡해짐 (여러 DB 상태 관리)


### 해결 방법 1: 이벤트 기반 분리
```kotlin
@Service
class ShortenUrlService(
    private val urlRepository: UrlRepository,
    private val eventPublisher: ApplicationEventPublisher  // Spring 이벤트
) {
    @Transactional
    fun createShortUrl(req: ShortUrlRequest): ShortUrlResponse {
        val url = urlRepository.save(UrlEntity(...))
        
        // ✅ 이벤트 발행
        eventPublisher.publishEvent(UrlCreatedEvent(url.id))
        
        return ShortUrlResponse(...)
    }
}

@Component
class AnalyticsEventListener(
    private val analyticsRepository: AnalyticsRepository
) {
    @Async
    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    fun handleUrlCreated(event: UrlCreatedEvent) {
        // ✅ 별도 트랜잭션에서 실행
        // ✅ 실패해도 URL 생성에 영향 없음
        analyticsRepository.save(AnalyticsEvent(event.urlId))
    }
}

data class UrlCreatedEvent(val urlId: Long)
```

```
@Service
class UserService(
    private val userRepository: UserRepository,
    private val shortenUrlService: ShortenUrlService  // 순환 참조!
) {
    fun getUserWithUrls(userId: Long): UserWithUrls {
        val user = userRepository.findById(userId)
        val urls = shortenUrlService.getUserUrls(userId)  // 다시 ShortenUrlService 호출
        return UserWithUrls(user, urls)
    }
```

### 해결방법2: Facade계층 도입
```kotlin
@Service
class ShortenUrlService(
    private val urlRepository: UrlRepository,
    private val randomShortCodeGenerator: RandomShortCodeGenerator
) {
    fun createShortUrl(req: ShortUrlRequest): UrlEntity {
        // 단일 책임: URL 생성만
        val shortCode = randomShortCodeGenerator.generate()
        return urlRepository.save(UrlEntity(shortCode = shortCode, ...))
    }
    
    fun getUserUrls(userId: Long): List<UrlEntity> {
        return urlRepository.findByUserId(userId)
    }
}

@Service
class UserService(
    private val userRepository: UserRepository
) {
    fun getUser(userId: Long): User {
        // 단일 책임: User 조회만
        return userRepository.findById(userId).orElseThrow()
    }
}

@Service
class UserUrlFacadeService(
    private val userService: UserService,
    private val shortenUrlService: ShortenUrlService
) {
    fun createShortUrlForUser(userId: Long, req: ShortUrlRequest): ShortUrlResponse {
        // 여러 서비스 조율
        val user = userService.getUser(userId)
        val url = shortenUrlService.createShortUrl(req)
        
        return ShortUrlResponse(
            userId = user.id,
            userName = user.name,
            shortUrl = url.shortCode
        )
    }
}
```
